SELECT
c.record_id AS id,
c.titre,
c.message,
c.usage,
c.bureau,
c.description,
c.modele_defaut AS "modeleDefaut",
c.cree_par ->> 'name' AS "creePar",
c.derniere_modification ->> 'name' AS "modifiePar"
FROM configuration_sms c
WHERE NOT EXISTS (SELECT 1 FROM chat.employes e WHERE lower(e.email) = lower({{appsmith.user.email}}))
   OR EXISTS (SELECT 1 FROM chat.employes e WHERE lower(e.email) = lower({{appsmith.user.email}}) AND (e.role = 'direction' OR e.bureau = c.bureau))
   OR (
     c.bureau IS NULL
     AND EXISTS (
       SELECT 1
       FROM chat.employes e
       WHERE lower(e.email) = lower({{appsmith.user.email}})
         AND NOT EXISTS (
           SELECT 1
           FROM configuration_sms o
           WHERE o.bureau = e.bureau
             AND ((o.usage IS NOT NULL AND o.usage = c.usage)
               OR (o.modele_defaut IS NOT NULL AND o.modele_defaut = c.record_id))
         )
     )
   );
